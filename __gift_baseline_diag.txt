// Comprueba el descanso de 10h entre el turno nuevo de un dia y los dias adyacentes.
function _validarDescanso10h(empDias, monthKey, dia, e1n, s1n, e2n, s2n, etiqueta){
  var errores=[];
  var DESC_MIN=600; // 10h
  if(e1n===null)return errores; // ese dia libra
  var miEntrada=e1n;
  var miSalida=(e2n!==null&&s2n!==null)?s2n:s1n;
  if(miSalida<miEntrada)miSalida+=1440; // turno nocturno: salida al dia siguiente
  function turnoDe(d){
    var t=empDias[d];
    if(!_esTurnoNormalCambio(t))return null;
    var e1=_toMin(t.e1), s1=_toMin(t.s1);
    var e2=t.e2?_toMin(t.e2):null, s2=t.s2?_toMin(t.s2):null;
    var sal=(e2!==null&&s2!==null)?s2:s1;
    if(sal<e1)sal+=1440;
    return {entrada:e1, salida:sal};
  }
  var ant=turnoDe(dia-1);
  if(ant){
    var descAntes=(1440-ant.salida)+miEntrada;
    if(descAntes<DESC_MIN)errores.push((etiqueta||'')+'Menos de 10h con el dia anterior ('+(descAntes/60).toFixed(1)+'h)');
  }
  var sig=turnoDe(dia+1);
  if(sig){
    var descDespues=(1440-miSalida)+sig.entrada;
    if(descDespues<DESC_MIN)errores.push((etiqueta||'')+'Menos de 10h con el dia siguiente ('+(descDespues/60).toFixed(1)+'h)');
  }
  return errores;
}

function _validarLibreSemana(empDias, monthKey, dia, e1n, etiqueta){
  var idx=_GRUPO_MONTHS.indexOf(monthKey);
  var fecha=new Date(_cambioYear,idx,dia);
  var dow=(fecha.getDay()+6)%7;
  var lunes=new Date(fecha.getTime()-dow*86400000);
  var diasPublicados=0, libres=0;
  for(var i=0;i<7;i++){
    var f=new Date(lunes.getTime()+i*86400000);
    var d=f.getDate();

    if(mk!==monthKey)continue; // dia de otro mes = sin publicar

      if(t===undefined||t===null){continue;} // sin publicar, podria ser libre
      diasPublicados++;
      trabaja=_esTurnoNormalCambio(t);
    }
    if(!trabaja)libres++;
  }
  // Solo falla si todos los dias publicados son trabajados (ninguno libre)
  // Si hay dias sin publicar, podrian ser el dia libre
  if(diasPublicados===7&&libres===0)return[(etiqueta||'')+'Sin dia libre esa semana'];

// Valida descanso semanal de 36h: entre fin del ultimo dia trabajado y inicio del siguiente bloque
function _validarDescansoSemanal(empDias, monthKey, dia, e1n, s1n, e2n, s2n, etiqueta){
  var errores=[];
  var DESC_SEM=2160; // 36h en minutos
  var idx=_GRUPO_MONTHS.indexOf(monthKey);
  var maxDia=new Date(_cambioYear,idx+1,0).getDate();

  function entradaDe(d){
    var t=d===dia?{e1:e1n!==null?_fmtMin(e1n):null}:(empDias[d]||null);
    if(d!==dia&&!_esTurnoNormalCambio(t))return null;
    return t&&t.e1?_toMin(t.e1):null;
  }
  function salidaDe(d){
    if(d===dia){
      if(e1n===null)return null;
      return (e2n!==null&&s2n!==null)?s2n:s1n;
    }
    var t=empDias[d]||null;
    if(!_esTurnoNormalCambio(t))return null;
    var s=t.e2&&t.s2?_toMin(t.s2):_toMin(t.s1);
    return s;
  }
  function trabajaEn(d){return entradaDe(d)!==null;}

  // Buscar bloques de trabajo separados por al menos 1 libre
  // Para cada transicion: ultimo_trabajo → libres → primer_trabajo
  for(var d=1;d<=maxDia;d++){
    if(!trabajaEn(d))continue;
    // Buscar el siguiente dia de trabajo tras un gap de libres
    var salD=salidaDe(d);
    // Avanzar hasta encontrar gap (libre) y luego siguiente trabajo
    var d2=d+1;
    var tieneLibre=false;
    while(d2<=maxDia&&!trabajaEn(d2)){tieneLibre=true;d2++;}
    if(d2>maxDia||!tieneLibre)continue;
    var entD2=entradaDe(d2);
    if(salD===null||entD2===null)continue;
    // Calcular horas entre salida de d y entrada de d2
    var difDias=d2-d;
    var descTotal=difDias*1440-salD+entD2;
    if(descTotal<DESC_SEM){
      errores.push((etiqueta||'')+'Descanso semanal <36h entre dia '+d+' y dia '+d2+' ('+(descTotal/60).toFixed(1)+'h)');
    }
  }
  return errores;
}
