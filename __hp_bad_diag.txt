function _grupoResumenImportadosCompaneros(map){

    Object.keys(emp.meses||{}).forEach(function(mk){
      if(!res[mk])res[mk]={total:0,cats:{}};
      res[mk].total++;
      res[mk].cats[cat]=(res[mk].cats[cat]||0)+1;
    });
  });
  var meses=Object.keys(res).sort(function(a,b){return _GRUPO_MONTHS.indexOf(a)-_GRUPO_MONTHS.indexOf(b);});
  if(!meses.length)return 'Sin meses guardados';
  return meses.map(function(mk){
    var cats=Object.keys(res[mk].cats).sort().map(function(c){return c+'('+res[mk].cats[c]+')';}).join(', ');
    return mk+': '+res[mk].total+' empleados · '+cats;
  }).join('\n');
}
function _grupoBorrarMesCategoriaCompaneros(map,monthKey,categoriaFiltro){

  Object.keys(map||{}).forEach(function(k){
    var emp=map[k]||{};
    if(!emp.meses||!emp.meses[monthKey])return;
    if(catNorm){
      var empCat=_normNombreGrupo(emp.categoria||'');
      if(empCat.indexOf(catNorm)<0&&catNorm.indexOf(empCat)<0)return;
    }

    delete emp.meses[monthKey];
    empleadosAfectados++;
    if(!Object.keys(emp.meses||{}).length)delete map[k];
    else map[k]=emp;
  });

}

function gestionarImportacionesCompaneros(){
  var map=_grupoLeerMapCompaneros();
  if(!Object.keys(map).length){alert('No hay importaciones guardadas en Compañeros por día.');return;}
  var resumen=_grupoResumenImportadosCompaneros(map);
  var msg='Gestión de importaciones de Compañeros por día\n\n'+
    'Escribe una opción:\n'+
    'ULTIMA = deshacer la última importación hecha desde ahora en adelante\n'+
    'TODO = borrar todo Compañeros por día\n'+

    'JULIO/FTP OPERARIO = borrar solo esa categoría de ese mes\n\n'+
    'Guardado ahora:\n'+resumen;
  var resp=prompt(msg,'');
  if(resp===null)return;
  resp=String(resp||'').trim();
  if(!resp)return;
  var up=resp.toUpperCase();
  if(up==='ULTIMA'||up==='DESHACER'){
    var backup=null;
    try{backup=JSON.parse(APP_STORAGE.getItem(_GRUPO_KEY+'_map_backup')||'null');}catch(e){backup=null;}
    if(!backup||!backup.before){alert('No hay copia anterior disponible. La copia se guarda a partir de las próximas importaciones.');return;}

    if(_grupoGuardarMapCompaneros(backup.before)){
      try{APP_STORAGE.removeItem(_GRUPO_KEY+'_map_backup');}catch(e){}
      _grupoRefrescarTrasGestion();
    }
    return;
  }
  if(up==='TODO'||up==='TODOS'||up==='ALL'){
    if(!confirm('Se borrarán todos los datos importados en Compañeros por día.'))return;
    APP_STORAGE.removeItem(_GRUPO_KEY+'_map');
    APP_STORAGE.removeItem(_GRUPO_KEY+'_map_backup');
    _grupoRefrescarTrasGestion();
    return;
  }
  var totalEmp=0,totalDias=0,fallidos=[];
  resp.split(/[;,\n]+/).forEach(function(part){

    var piezas=part.split(/[\/|]+/);
    var monthKey=_grupoNormalizarMesInput(piezas[0]);
    var categoria=piezas.slice(1).join('/').trim();

  }
}

// Códigos de día que nunca son un nombre real, por si se han colado como "nombre"
// debido a una importación corrupta (columnas desalineadas).
var _DIAS_CODIGO_BASURA=['DL','LIBRE','DIALIBRE','DESCANSO','DESCANSOSEMANA','MA','FM','FY','EXC'];
function _esNombreBasuraGrupo(nombre){
  var n=String(nombre||'').trim();
  if(!n)return true;
  if(/\d/.test(n))return true; // un nombre real de persona nunca lleva dígitos
  if(/[()]/.test(n))return true; // un nombre real no lleva paréntesis (p.ej. "(EXC)")
  var up=n.toUpperCase().replace(/[\s.]/g,'');
  return _DIAS_CODIGO_BASURA.indexOf(up)>=0;
}
function _normNombreGrupo(nombre){
  return String(nombre||'').toUpperCase().normalize('NFD').replace(/[\u0300-\u036f]/g,'').replace(/\s+/g,' ').trim();
}
